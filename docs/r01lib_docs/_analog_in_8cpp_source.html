55void AnalogIn::resolve_pin( int pin )
56{
57 for ( size_t i = 0; i < sizeof( s_pins ) / sizeof( s_pins[0] ); i++ )
58 {
59 if ( s_pins[ i ].pin == pin )
60 {
61 _input_positive = (uint8_t)s_pins[ i ].input_positive;
62 _channel_id = (uint8_t)( i + 1 ); // dedicated LPADC CMD1..CMD4
63 _pin = pin;
64
65 port_pin_config_t cfg = {
66 kPORT_PullDisable,
67 kPORT_LowPullResistor,
68 kPORT_FastSlewRate,
69 kPORT_PassiveFilterDisable,
70 kPORT_OpenDrainDisable,
71 kPORT_LowDriveStrength,
72 kPORT_NormalDriveStrength,
73 kPORT_MuxAlt0, // ADC0_An
74 kPORT_InputBufferDisable, // required for analog function
75 kPORT_InputNormal,
76 kPORT_UnlockRegister
77 };
78 PORT_SetPinConfig( s_pins[ i ].port, s_pins[ i ].port_pin, &cfg );
79 return;
80 }
81 }
82
83 _input_positive = 0;
84 _channel_id = 0;
85 _pin = -1;
86}

88void AnalogIn::_acquire_peripheral( void )
89{
90 if ( _instance_count == 0 )
91 {
92 CLOCK_EnableClock( kCLOCK_GateADC0 );
93 RESET_ReleasePeripheralReset( kADC0_RST_SHIFT_RSTn );
94
95 // ADCK (conversion clock) source — without this, LPADC_Init()
96 // succeeds but no conversion/calibration cycle ever completes and
97 // LPADC_DoAutoCalibration() spins forever polling GCC[0].RDY.
98 CLOCK_SetClockDiv( kCLOCK_DivADC0, 1u );
99 CLOCK_AttachClk( kFRO12M_to_ADC0 );
100
101 lpadc_config_t cfg;
102 LPADC_GetDefaultConfig( &cfg );
103 cfg.enableAnalogPreliminary = true;
104 cfg.referenceVoltageSource = kLPADC_ReferenceVoltageAlt3; // VDDA (FRDM-MCXA153)
105 cfg.powerLevelMode = kLPADC_PowerLevelAlt4; // highest accuracy
106 LPADC_Init( ADC0, &cfg );
107
108#if defined( FSL_FEATURE_LPADC_HAS_CTRL_CAL_REQ ) && FSL_FEATURE_LPADC_HAS_CTRL_CAL_REQ
109 if ( !_calibrated )
110 {
111 LPADC_DoAutoCalibration( ADC0 );
112 _calibrated = true;
113 }
114#endif
115 }
116
117 _instance_count++;
118}

120void AnalogIn::_release_peripheral( void )
121{
122 _instance_count--;
123
124 if ( _instance_count == 0 )
125 {
126 LPADC_Deinit( ADC0 );
127 _calibrated = false; // re-calibrate on the next acquire after a full deinit
128 }
129}

131AnalogIn::AnalogIn( int pin )
132 : Obj( true ), _channel_id( 0 ), _input_positive( 0 ), _pin( -1 )
133{
134 resolve_pin( pin );
135
136 if ( -1 == _pin )
137 {
138 panic( "AnalogIn: unsupported analog pin" );
139 return;
140 }
141
142 _acquire_peripheral();
143
144 lpadc_conv_command_config_t cmd;
145 LPADC_GetDefaultConvCommandConfig( &cmd );
146 cmd.channelNumber = _input_positive;
147 cmd.conversionResolutionMode = kLPADC_ConversionResolutionStandard; // 12-bit
148 LPADC_SetConvCommandConfig( ADC0, _channel_id, &cmd );
149}

159uint16_t AnalogIn::sample_raw( void )
160{
161 if ( -1 == _pin )
162 return 0;
163
164 lpadc_conv_trigger_config_t trig;
165 LPADC_GetDefaultConvTriggerConfig( &trig );
166 trig.targetCommandId = _channel_id;
167 trig.enableHardwareTrigger = false;
168 LPADC_SetConvTriggerConfig( ADC0, 0u, &trig );
169
170 LPADC_DoResetFIFO( ADC0 );
171 LPADC_DoSoftwareTrigger( ADC0, 1u ); // trigger0 mask
172
173 lpadc_conv_result_t result;
174 while ( !LPADC_GetConvResult( ADC0, &result ) )
175 ;
176
177 return (uint16_t)( ( result.convValue >> 3u ) & 0x0FFFu );
178}
